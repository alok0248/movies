import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import android.util.Base64
import java.nio.charset.StandardCharsets

fun sendLog(message: String, level: String = "info", tag: String = "", data: JSONObject? = null) {
    val url = "{{ request.scheme }}://{{ request.get_host }}/api/android-apps/{{ selected_app.slug }}/log/"
    val credentials = "{{ selected_app.access_username }}:{{ selected_app.access_password }}"
    val auth = "Basic " + Base64.encodeToString(credentials.toByteArray(StandardCharsets.UTF_8), Base64.NO_WRAP)

    val body = JSONObject().apply {
        put("message", message)
        put("level", level)
        put("tag", tag)
        put("build_identifier", BuildConfig.VERSION_CODE.toString())
        put("device_model", android.os.Build.MODEL)
        put("os_version", android.os.Build.VERSION.RELEASE)
        if (data != null) put("data", data)
    }

    val request = Request.Builder()
        .url(url)
        .addHeader("Authorization", auth)
        .post(body.toString().toRequestBody("application/json".toMediaType()))
        .build()

    OkHttpClient().newCall(request).execute().close()
}

// Usage:
sendLog("App started successfully", level = "info", tag = "AppLifecycle")
sendLog("API key failed to load", level = "error", tag = "Auth", data = JSONObject().put("attempt", 3))
